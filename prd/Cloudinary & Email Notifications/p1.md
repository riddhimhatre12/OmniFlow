Set up Cloudinary integration for OmniFlow to handle all media uploads 
(task attachments, user avatars, workspace logos).

Tech: Cloudinary, Convex, React 19

---

## STEP 1: Create Cloudinary Account & Get Credentials

1. Sign up at https://cloudinary.com (free tier available)
2. Go to Dashboard → Settings
3. Collect these credentials:
   - Cloud Name (e.g., "dxyz1a2bc")
   - API Key (e.g., "123456789...")
   - API Secret (e.g., "abc123def...")
   - Unsigned Upload Preset name (create new if needed)

4. Create Upload Presets (Dashboard → Settings → Upload):
   - Preset 1: "omniflow_attachments"
     * Folder: /omniflow/attachments
     * Type: Unsigned
     * Allow formats: jpg, jpeg, png, pdf, doc, docx, xls, xlsx, zip
     * Max file size: 50 MB
   
   - Preset 2: "omniflow_avatars"
     * Folder: /omniflow/avatars
     * Type: Unsigned
     * Allow formats: jpg, jpeg, png, gif, webp
     * Max file size: 5 MB
     * Transformations: Auto-crop faces (optional)
   
   - Preset 3: "omniflow_workspace_logos"
     * Folder: /omniflow/logos
     * Type: Unsigned
     * Allow formats: jpg, jpeg, png, svg, webp
     * Max file size: 2 MB

## STEP 4: Create Cloudinary Backend Functions

import { v } from "convex/values";
import { mutation, query } from "./_generated/server";
import { Id } from "./_generated/dataModel";

// Generate a signed upload token for client-side upload
export const generateUploadSignature = mutation({
  args: {
    uploadPreset: v.string(),
    folder: v.optional(v.string()),
  },
  handler: async (ctx, args) => {
    // Validate upload preset
    const validPresets = [
      process.env.CLOUDINARY_PRESET_ATTACHMENTS,
      process.env.CLOUDINARY_PRESET_AVATARS,
      process.env.CLOUDINARY_PRESET_LOGOS,
    ];
    
    if (!validPresets.includes(args.uploadPreset)) {
      throw new Error("Invalid upload preset");
    }

    // For unsigned uploads, no signature needed
    // Return cloud name and preset
    return {
      cloudName: process.env.CLOUDINARY_CLOUD_NAME,
      uploadPreset: args.uploadPreset,
      folder: args.folder || "",
    };
  },
});

// Process upload response and save metadata to database
export const saveAttachmentMetadata = mutation({
  args: {
    taskId: v.id("tasks"),
    workspaceId: v.id("workspaces"),
    userId: v.id("users"),
    fileName: v.string(),
    cloudinaryPublicId: v.string(),
    cloudinaryUrl: v.string(),
    cloudinarySecureUrl: v.string(),
    cloudinaryFormat: v.string(),
    cloudinaryResourceType: v.string(),
    cloudinaryBytes: v.number(),
    cloudinaryVersion: v.number(),
    description: v.optional(v.string()),
  },
  handler: async (ctx, args) => {
    // Verify user has workspace access
    const workspaceMember = await ctx.db
      .query("workspace_members")
      .withIndex("by_workspace_and_user", (q) =>
        q.eq("workspaceId", args.workspaceId).eq("userId", args.userId)
      )
      .first();

    if (!workspaceMember) {
      throw new Error("User does not have access to this workspace");
    }

    // Verify task belongs to workspace
    const task = await ctx.db.get(args.taskId);
    if (!task || task.workspaceId !== args.workspaceId) {
      throw new Error("Task not found in workspace");
    }

    // Save attachment metadata
    const attachmentId = await ctx.db.insert("task_attachments", {
      taskId: args.taskId,
      workspaceId: args.workspaceId,
      uploadedBy: args.userId,
      fileName: args.fileName,
      fileType: args.cloudinaryFormat,
      fileSize: args.cloudinaryBytes,
      cloudinaryPublicId: args.cloudinaryPublicId,
      cloudinaryUrl: args.cloudinaryUrl,
      cloudinarySecureUrl: args.cloudinarySecureUrl,
      cloudinaryFormat: args.cloudinaryFormat,
      cloudinaryResourceType: args.cloudinaryResourceType,
      cloudinaryBytes: args.cloudinaryBytes,
      cloudinaryVersion: args.cloudinaryVersion,
      description: args.description || "",
      uploadedAt: Date.now(),
      isDeleted: false,
    });

    // Log activity
    await ctx.db.insert("activity_feed", {
      workspaceId: args.workspaceId,
      userId: args.userId,
      action: "file_uploaded",
      resourceType: "task",
      resourceId: args.taskId,
      resourceName: args.fileName,
      description: `${(await ctx.db.get(args.userId))?.name} uploaded file "${args.fileName}" to task`,
      createdAt: Date.now(),
    });

    return attachmentId;
  },
});

// Save avatar metadata
export const saveAvatarMetadata = mutation({
  args: {
    userId: v.id("users"),
    cloudinaryPublicId: v.string(),
    cloudinaryUrl: v.string(),
    cloudinarySecureUrl: v.string(),
    cloudinaryFormat: v.string(),
    cloudinaryBytes: v.number(),
  },
  handler: async (ctx, args) => {
    // Delete old avatar if exists
    const oldAvatar = await ctx.db
      .query("user_avatars")
      .withIndex("by_userId", (q) => q.eq("userId", args.userId))
      .first();

    if (oldAvatar) {
      // Optionally delete from Cloudinary using API
      await ctx.db.delete(oldAvatar._id);
    }

    // Save new avatar
    await ctx.db.insert("user_avatars", {
      userId: args.userId,
      cloudinaryPublicId: args.cloudinaryPublicId,
      cloudinaryUrl: args.cloudinaryUrl,
      cloudinarySecureUrl: args.cloudinarySecureUrl,
      format: args.cloudinaryFormat,
      bytes: args.cloudinaryBytes,
      uploadedAt: Date.now(),
    });

    // Update user profile with avatar URL
    await ctx.db.patch(args.userId, {
      avatar: args.cloudinarySecureUrl,
    });
  },
});

// Save workspace logo metadata
export const saveWorkspaceLogoMetadata = mutation({
  args: {
    workspaceId: v.id("workspaces"),
    userId: v.id("users"),
    cloudinaryPublicId: v.string(),
    cloudinaryUrl: v.string(),
    cloudinarySecureUrl: v.string(),
    cloudinaryFormat: v.string(),
    cloudinaryBytes: v.number(),
  },
  handler: async (ctx, args) => {
    // Verify user is workspace owner/admin
    const member = await ctx.db
      .query("workspace_members")
      .withIndex("by_workspace_and_user", (q) =>
        q.eq("workspaceId", args.workspaceId).eq("userId", args.userId)
      )
      .first();

    if (!member) {
      throw new Error("User not in workspace");
    }

    const role = await ctx.db.get(member.roleId);
    if (role?.name !== "Owner" && role?.name !== "Admin") {
      throw new Error("Only Owner/Admin can change workspace logo");
    }

    // Delete old logo if exists
    const oldLogo = await ctx.db
      .query("workspace_logos")
      .withIndex("by_workspaceId", (q) =>
        q.eq("workspaceId", args.workspaceId)
      )
      .first();

    if (oldLogo) {
      await ctx.db.delete(oldLogo._id);
    }

    // Save new logo
    await ctx.db.insert("workspace_logos", {
      workspaceId: args.workspaceId,
      cloudinaryPublicId: args.cloudinaryPublicId,
      cloudinaryUrl: args.cloudinaryUrl,
      cloudinarySecureUrl: args.cloudinarySecureUrl,
      format: args.cloudinaryFormat,
      bytes: args.cloudinaryBytes,
      uploadedAt: Date.now(),
      uploadedBy: args.userId,
    });

    // Update workspace with logo URL
    await ctx.db.patch(args.workspaceId, {
      avatar: args.cloudinarySecureUrl,
    });
  },
});

// Query to get attachment download URL with transformations
export const getAttachmentUrl = query({
  args: {
    attachmentId: v.id("task_attachments"),
    width: v.optional(v.number()),
    height: v.optional(v.number()),
    format: v.optional(v.string()),
  },
  handler: async (ctx, args) => {
    const attachment = await ctx.db.get(args.attachmentId);

    if (!attachment || attachment.isDeleted) {
      throw new Error("Attachment not found");
    }

    // Build Cloudinary URL with optional transformations
    let url = attachment.cloudinarySecureUrl;

    if (args.width || args.height || args.format) {
      const transformations = [];

      if (args.width && args.height) {
        transformations.push(`w_${args.width},h_${args.height},c_fill`);
      } else if (args.width) {
        transformations.push(`w_${args.width},c_scale`);
      } else if (args.height) {
        transformations.push(`h_${args.height},c_scale`);
      }

      if (args.format) {
        transformations.push(`f_${args.format}`);
      }

      // Insert transformations into URL
      // Original: https://res.cloudinary.com/cloud/image/upload/v123/path.jpg
      // Becomes: https://res.cloudinary.com/cloud/image/upload/w_400,h_300,c_fill/v123/path.jpg
      url = url.replace("/upload/", `/upload/${transformations.join(",")}/`);
    }

    return url;
  },
});
