import { scheduled } from "./_generated/server";
import { v } from "convex/values";

// Scheduled function: runs daily at 9 AM (or user's preferred time)
export const sendDailyDigests = scheduled("0 9 * * *", async (ctx) => {
  // Get all users who have email notifications enabled
  const users = await ctx.db
    .query("users")
    .collect();

  for (const user of users) {
    // Get user's notification preferences
    const prefs = await ctx.db
      .query("notification_preferences")
      .filter((q) => q.eq(q.field("userId"), user._id))
      .first();

    if (!prefs || !prefs.emailNotifications) {
      continue; // Skip if user disabled email notifications
    }

    // Get user's tasks
    const tasksAssigned = await ctx.db
      .query("tasks")
      .filter((q) => q.eq(q.field("assigneeId"), user._id))
      .collect();

    // Get tasks due today
    const today = new Date();
    today.setHours(0, 0, 0, 0);
    const tomorrowStart = new Date(today);
    tomorrowStart.setDate(tomorrowStart.getDate() + 1);

    const dueTodayTasks = tasksAssigned.filter(
      (task) =>
        task.dueDate >= today.getTime() &&
        task.dueDate < tomorrowStart.getTime()
    );

    // Get upcoming sprints (due in next 14 days)
    const upcomingDeadlines = await ctx.db
      .query("sprints")
      .filter((q) => q.eq(q.field("status"), "active"))
      .collect();

    // Get mentions (comments with @username)
    const mentions = await ctx.db
      .query("task_comments")
      .filter((q) =>
        q.and(
          q.eq(q.field("userId"), user._id),
          q.gte(q.field("createdAt"), Date.now() - 24 * 60 * 60 * 1000)
        )
      )
      .collect();

    // Only send if there's something to report
    if (tasksAssigned.length === 0 && dueTodayTasks.length === 0 && mentions.length === 0) {
      continue;
    }

    // Format data for email
    const taskList = tasksAssigned.slice(0, 5).map((task) => {
      const project = await ctx.db.get(task.projectId);
      return {
        title: task.title,
        projectName: project.name,
        dueDate: formatDate(task.dueDate),
      };
    });

    // Send digest email
    await ctx.runAction(api.mailer.sendDailyDigestEmail, {
      userEmail: user.email,
      userName: user.name,
      tasksAssigned: taskList,
      dueTodayCount: dueTodayTasks.length,
      upcomingDeadlines: upcomingDeadlines
        .filter((sprint) => sprint.endDate < Date.now() + 14 * 24 * 60 * 60 * 1000)
        .map((sprint) => ({
          sprintName: sprint.name,
          daysUntilEnd: Math.ceil(
            (sprint.endDate - Date.now()) / (24 * 60 * 60 * 1000)
          ),
        })),
      mentionCount: mentions.length,
    });
  }
});

// Helper function to format dates
function formatDate(timestamp: number): string {
  return new Date(timestamp).toLocaleDateString("en-US", {
    year: "numeric",
    month: "short",
    day: "numeric",
  });
}

## **Implementation Notes for Milestone 16:**

1. **Email Provider Choice:**
   - Resend: Easier setup, good for startups, free tier 100 emails/day
   - SendGrid: More features, better support, paid plans
   - Both integrate easily with Convex

2. **Preference Checks:**
   - Always check `notification_preferences.emailNotifications` before sending
   - And check specific notification type toggle (taskAssignedEnabled, etc.)

3. **Unsubscribe Link:**
   - Add to footer of every email: link to /preferences to manage notifications
   - Or simple query param: `?unsubscribe=true` in preference page

4. **Testing:**
   - Use Resend test API key for development
   - Or SendGrid sandbox mode
   - Don't send real emails to test accounts

5. **Rate Limiting:**
   - Add delay between emails in cron jobs to avoid hitting API limits
   - Use batch sending if provider supports

6. **Logging:**
   - Log all email sends to activity feed for audit trail
   - Save email status: "sent", "failed", "bounced"

7. **Email Design:**
   - Mobile-responsive HTML templates
   - Clear CTAs (Call-to-Actions)
   - Branding (logo, colors)
   - Unsubscribe footer
